1. select "Payment Method",sum("Total Spent") over( partition by "Payment Method") from cafesales
2. select "Item", sum("Total Spent") as sum_sales from cafesales group by "Item" order by "sum_sales" desc limit 5
3. select "Item", sum("Quantity") as total_quantity from cafesales group by ("Item") order by(total_quantity) desc
4. with item_price_per_unit as 
(select "Item", avg("Price Per Unit") as avg_price_per_unit from cafesales group by "Item")

select * from item_price_per_unit order by avg_price_per_unit desc
5. select "Location", sum("Total Spent") from cafesales group by "Location"
6. with spends_by_location as 
(select "Location", sum("Total Spent") as total_spent from cafesales group by "Location")

select "Location" from spends_by_location order by total_spent
7. select "Transaction Date" from cafesales order by "Transaction Date" desc
8. select "Transaction Date", "Total Spent"  from cafesales order by "Total Spent" desc
9. select "Item", avg("Total Spent") as total_spent from cafesales group by "Item" order by total_spent desc
10. select * from cafesales where "Item" = 'Coffee'
11. explain analyze select * from cafesales where "Item" = 'Coffee'
